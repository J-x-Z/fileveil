cmake_minimum_required(VERSION 3.21)
project(FileVeilRealProjectAcceptance LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
if(WIN32)
  # Hosted Windows acceptance uses the static vcpkg Zlib triplet (zs.lib).
  set(ZLIB_USE_STATIC_LIBS ON)
endif()
find_package(ZLIB REQUIRED)
add_library(calculation STATIC calculation.cpp compression.c)
target_link_libraries(calculation PUBLIC ZLIB::ZLIB)
add_executable(fv-calculate calculate.cpp)
target_link_libraries(fv-calculate PRIVATE calculation)
add_executable(fv-compress compress.cpp)
target_link_libraries(fv-compress PRIVATE calculation)
enable_testing()
add_test(NAME sum COMMAND fv-calculate 1000)
set_tests_properties(sum PROPERTIES PASS_REGULAR_EXPRESSION "500500")
add_test(NAME compression COMMAND fv-compress 1000)
set_tests_properties(compression PROPERTIES PASS_REGULAR_EXPRESSION "roundtrip:1000")
